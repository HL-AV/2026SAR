    function searchData() {
        const name = document.getElementById('userName').value.trim();
        const birth = document.getElementById('userBirth').value.trim();
        const resultDiv = document.getElementById('result');
        
        resultDiv.innerHTML = "";

        if(!name || !birth) {
            alert("이름과 생년월일을 모두 입력해주세요.");
            return;
        }

        // 이름과 생년월일이 모두 일치하는 데이터 필터링 (중복 포함)
        const matches = data.filter(person => person.name === name && person.birth === birth);

        if (matches.length > 0) {
            let html = `<h4>총 ${matches.length}건의 실적이 확인되었습니다.</h4>`;
            matches.forEach((item, index) => {
                html += `
                    <div class="record-card">
                        <p><strong>[데이터 ${index + 1}]</strong></p>
                        <p>근무일: <b>${item.days}일</b></p>
                        <p>매출: <b>${item.sales}원</b> (목표: ${item.target})</p>
                        <p>매출 달성률: <span class="highlight">${item.rate}</span></p>
                        <p>품목 판매수량: <b>${item.itemQty}개</b></p>
                        <p>품목 달성률: <b>${item.itemRate}</b></p>
                    </div>
                `;
            });
            resultDiv.innerHTML = html;
        } else {
            resultDiv.innerHTML = "<p style='text-align:center; color:#999;'>일치하는 데이터가 없습니다.</p>";
        }
    }
